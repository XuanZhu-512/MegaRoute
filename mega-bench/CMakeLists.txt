cmake_minimum_required(VERSION 3.20)
project(pcb_bench VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MR_SOURCE_FILES src/geometry.cpp src/spatial.cpp src/model.cpp src/drc.cpp src/router.cpp
    include/megaroute/core.hpp apps/main.cpp CMakeLists.txt include/megaroute/build_identity.hpp.in)
set(MR_SOURCE_DIGESTS "")
foreach(source IN LISTS MR_SOURCE_FILES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND MR_SOURCE_DIGESTS "${source}:${source_hash}\n")
endforeach()
string(SHA256 MR_SOURCE_FINGERPRINT "${MR_SOURCE_DIGESTS}")
configure_file(include/megaroute/build_identity.hpp.in generated/megaroute/build_identity.hpp @ONLY)
add_library(megaroute src/geometry.cpp src/spatial.cpp src/model.cpp src/drc.cpp src/router.cpp)
target_include_directories(megaroute PUBLIC include)
target_include_directories(megaroute SYSTEM PUBLIC third_party)
if(MSVC)
  target_compile_options(megaroute PRIVATE /W4)
else()
  target_compile_options(megaroute PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(megaroute_cli apps/main.cpp)
target_include_directories(megaroute_cli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(megaroute_cli PRIVATE megaroute)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE megaroute)
add_test(NAME core COMMAND core_tests)
add_executable(rule_semantics_tests tests/rule_semantics_tests.cpp)
target_link_libraries(rule_semantics_tests PRIVATE megaroute)
add_test(NAME rule_semantics COMMAND rule_semantics_tests)
add_executable(routing_regressions tests/routing_regressions.cpp)
target_link_libraries(routing_regressions PRIVATE megaroute)
add_test(NAME routing_regressions COMMAND routing_regressions)
add_executable(polygon_holes_tests tests/polygon_holes_tests.cpp)
target_link_libraries(polygon_holes_tests PRIVATE megaroute)
add_test(NAME polygon_holes COMMAND polygon_holes_tests)
add_executable(terminal_regions_tests tests/terminal_regions_tests.cpp)
target_link_libraries(terminal_regions_tests PRIVATE megaroute)
add_test(NAME terminal_regions COMMAND terminal_regions_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME baseline_repairs COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_baseline_repairs.py)
  add_test(NAME freerouting_adapter COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_freerouting.py)
  add_test(NAME routing_workers COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_routing_improvements.py)
  add_test(NAME megaroute_defaults COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_megaroute_defaults.py)
  add_test(NAME native_geometry_export COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_geometry_export.py)
  add_test(NAME routing_adapter COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_adapt.py)
  add_test(NAME tracemaker_backend COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tracemaker.py)
  add_test(NAME architecture_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_architecture.py)
  add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:megaroute_cli> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
